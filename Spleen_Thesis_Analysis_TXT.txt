get_top_DEGs_SP <- function(de_table, n = 20, direction = c("up", "down")) {
  direction <- match.arg(direction)
  if (direction == "up") {
    genes <- rownames(de_table)[de_table$group == "up"]
    genes <- genes[order(-de_table[genes, "avg_log2FC"])]
  } else {
    genes <- rownames(de_table)[de_table$group == "down"]
    genes <- genes[order(de_table[genes, "avg_log2FC"])]
  }
  head(genes, n)
}

make_GO_heatmap_matrix_SP <- function(go_object, log2FC_df, top_n = 5) {
  go_df <- as.data.frame(go_object)
  if (nrow(go_df) == 0) return(NULL)
  
  go_df <- go_df %>%
    dplyr::rowwise() %>%
    dplyr::mutate(mean_log2FC = mean(log2FC_df$avg_log2FC[log2FC_df$gene %in% strsplit(geneID, "/")[[1]]], na.rm = TRUE)) %>%
    dplyr::ungroup() %>%
    dplyr::filter(!is.na(mean_log2FC))
  
  combined <- dplyr::bind_rows(
    go_df %>% dplyr::arrange(dplyr::desc(mean_log2FC)) %>% dplyr::slice_head(n = top_n),
    go_df %>% dplyr::arrange(mean_log2FC) %>% dplyr::slice_head(n = top_n)
  ) %>%
    dplyr::distinct(Description, .keep_all = TRUE)
  
  matrix(combined$mean_log2FC, nrow = nrow(combined), ncol = 1, dimnames = list(combined$Description, ""))
}

make_GO_direction_plot_SP <- function(go_object, de_table, plot_title, top_n = 10) {
  
  GO_term_genes <- as.data.frame(go_object) %>%
    dplyr::select(ID, Description, p.adjust, geneID) %>%
    tidyr::separate_rows(geneID, sep = "/") %>%
    dplyr::rename(gene = geneID) %>%
    dplyr::left_join(
      de_table %>%
        dplyr::select(gene, avg_log2FC, p_val_adj),
      by = "gene"
    ) %>%
    dplyr::mutate(
      direction = dplyr::case_when(
        avg_log2FC > 0 ~ "Up in KO",
        avg_log2FC < 0 ~ "Up in WT / Down in KO",
        TRUE ~ "No change"
      )
    )
  
  GO_direction_summary <- GO_term_genes %>%
    dplyr::group_by(Description) %>%
    dplyr::summarise(
      n_up_KO = sum(avg_log2FC > 0, na.rm = TRUE),
      n_up_WT = sum(avg_log2FC < 0, na.rm = TRUE),
      total = n_up_KO + n_up_WT,
      percent_up_KO = 100 * n_up_KO / total,
      percent_up_WT = 100 * n_up_WT / total,
      .groups = "drop"
    ) %>%
    dplyr::mutate(
      majority_direction = dplyr::case_when(
        percent_up_KO >= 70 ~ "mostly up in KO",
        percent_up_WT >= 70 ~ "mostly up in WT",
        TRUE ~ "mixed"
      )
    )
  
  top_terms <- as.data.frame(go_object) %>%
    dplyr::arrange(p.adjust) %>%
    dplyr::slice_head(n = top_n) %>%
    dplyr::pull(Description)
  
  GO_direction_plot <- GO_direction_summary %>%
    dplyr::filter(Description %in% top_terms) %>%
    tidyr::pivot_longer(
      cols = c(n_up_KO, n_up_WT),
      names_to = "direction",
      values_to = "gene_count"
    ) %>%
    dplyr::mutate(
      direction = dplyr::recode(
        direction,
        n_up_KO = "Up in KO",
        n_up_WT = "Up in WT / Down in KO"
      )
    )
  
  ggplot(
    GO_direction_plot,
    aes(
      x = gene_count,
      y = forcats::fct_reorder(Description, gene_count, .fun = sum),
      fill = direction
    )
  ) +
    geom_col(position = "fill") +
    scale_x_continuous(labels = scales::percent) +
    labs(
      title = plot_title,
      x = "Proportion of genes in GO term",
      y = "GO term",
      fill = "Gene direction"
    ) +
    theme_classic()
}

# GO enrichment function for CD4/CD8 subsets
###############################

run_GO_subset_SP <- function(sig_df, prefix) {
  
  genes <- sig_df$gene
  
  mapped <- bitr(
    genes,
    fromType = "SYMBOL",
    toType = "ENSEMBL",
    OrgDb = org.Mm.eg.db
  ) %>%
    dplyr::distinct(SYMBOL, .keep_all = TRUE)
  
  GO_BP <- enrichGO(mapped$ENSEMBL, OrgDb = org.Mm.eg.db, keyType = "ENSEMBL",
                    ont = "BP", pvalueCutoff = 0.05, pAdjustMethod = "BH", readable = TRUE)
  
  GO_MF <- enrichGO(mapped$ENSEMBL, OrgDb = org.Mm.eg.db, keyType = "ENSEMBL",
                    ont = "MF", pvalueCutoff = 0.05, pAdjustMethod = "BH", readable = TRUE)
  
  GO_CC <- enrichGO(mapped$ENSEMBL, OrgDb = org.Mm.eg.db, keyType = "ENSEMBL",
                    ont = "CC", pvalueCutoff = 0.05, pAdjustMethod = "BH", readable = TRUE)
  
  list(BP = GO_BP, MF = GO_MF, CC = GO_CC)
}

library(enrichplot)

library(ggplot2)

###############################
# Fold-change vector
# avg_log2FC > 0 = higher in KO
# avg_log2FC < 0 = higher in WT / down in KO
###############################

gene_fc_SP <- Deq_all_MAST_top_SP$avg_log2FC
names(gene_fc_SP) <- Deq_all_MAST_top_SP$gene

gene_fc_binary_SP <- ifelse(gene_fc_SP > 0, 1, -1)
names(gene_fc_binary_SP) <- names(gene_fc_SP)

binary_colour_scale_SP <- scale_color_gradient2(
  low = "blue",
  mid = "white",
  high = "red",
  midpoint = 0,
  breaks = c(-1, 1),
  labels = c("Down in KO", "Up in KO"),
  name = "Direction"
)
